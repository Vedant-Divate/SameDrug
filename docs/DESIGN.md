import hashlib, httpx

from pathlib import Path

from samedrug.pipeline.sources import NPPA\_CEILING\_URL, JAP\_PRICELIST\_URL, USER\_AGENT

def fetch(url: str, dest: Path) -> Path:

with httpx.Client(headers={"User-Agent": USER\_AGENT}, timeout=60, follow\_redirects=True) as c:

with c.stream("GET", url) as r:

r.raise\_for\_status()

dest.parent.mkdir(parents=True, exist\_ok=True)

with open(dest, "wb") as f:

for chunk in r.iter\_bytes():

f.write(chunk)

return dest

def download\_all(data\_dir: Path) -> dict\[str, Path\]:

return {

"nppa": fetch(NPPA\_CEILING\_URL, data\_dir / "raw" / "nppa\_ceiling.pdf"),

"jap": fetch(JAP\_PRICELIST\_URL, data\_dir / "raw" / "jap\_pricelist.xlsx"),

\## Phase 3 — The Normalization Engine (Days 7–11) — \*the actual project\*

Everything else in this build is plumbing. This module is why the project exists: three functions that turn messy human strings into one canonical match key.

\*\*\`samedrug/pipeline/normalize.py\`:\*\*

\`\`\`python

import re

from dataclasses import dataclass

FORM\_MAP = {

"tab": "tablet", "tab.": "tablet", "tablet": "tablet", "tabets": "tablet",

"cap": "capsule", "capsule": "capsule", "caps": "capsule",

"inj": "injection", "injection": "injection", "vial": "injection",

"syr": "syrup", "syrup": "syrup", "susp": "suspension",

"suspension": "suspension", "drops": "drops", "dt": "dispersible tablet",

"er": "extended release", "xl": "extended release", "sr": "sustained release",

"cream": "cream", "gel": "gel", "ointment": "ointment",

}

UNIT\_TO\_MG = {"mcg": 0.001, "mg": 1.0, "g": 1000.0}

STRENGTH\_PART = re.compile(r"(\\d+(?:\\.\\d+)?)\\s\*(mcg|mg|g|iu)\\b", re.I)

COMBO\_SPLIT = re.compile(r"\[+/\]")

@dataclass(frozen=True)

class FormulationKey:

molecule: str # canonical: "Atorvastatin"

strength: str # canonical: "10mg" | combos: "500mg+125mg"

form: str # canonical: "tablet"

def key(self) -> str:

return f"{self.molecule}|{self.strength}|{self.form}"

def parse\_strength(s: str) -> str:

"""'10MG'→'10mg', '500+125 mg'→'500mg+125mg', '1000MCG'→'1mg'"""

parts = COMBO\_SPLIT.split(s)

out = \[\]

for p in parts:

m = STRENGTH\_PART.search(p)

if not m:

raise ValueError(f"unparseable strength: {s!r}")

val, unit = float(m\[1\]), m\[2\].lower()

if unit in UNIT\_TO\_MG:

mg = val \* UNIT\_TO\_MG\[unit\]

out.append(f"{mg:g}mg")

else: # IU stays as-is

out.append(f"{val:g}iu")

return "+".join(sorted(out, key=str))

MOLECULE\_ALIASES = {

"metformin hcl": "metformin", "metformin hydrochloride": "metformin",

"amoxycillin": "amoxicillin", "poractant alfa": "poractant alfa",

"s-(-)-amlodipine": "amlodipine", "levocetirizine dihydrochloride": "levocetirizine",

\# grows as the parser's reject-queue reveals more

}

def canonical\_molecule(raw: str) -> str:

s = re.sub(r"\\s\*(IP|BP|USP|/)\\b", " ", raw, flags=re.I)

s = re.sub(r"\\((.\*?)\\)", " ", s) # strip "(as enteric coated)" etc.

s = re.sub(r"\\s+", " ", s).strip().lower()

s = MOLECULE\_ALIASES.get(s, s)

return s

def canonical\_form(raw: str) -> str:

return FORM\_MAP.get(raw.strip().lower(), raw.strip().lower())

\*\*\`samedrug/app/main.py\`:\*\*

\`\`\`python

from fastapi import FastAPI, HTTPException, Query

from fastapi.responses import JSONResponse

from samedrug.db import search\_formulations, get\_key

app = FastAPI(title="SameDrug", version="0.1.0")

@app.get("/api/search")

def search(q: str = Query(min\_length=2)):

\# q = molecule OR brand; rapidfuzz match over molecule+brand index

return {"results": search\_formulations(q)}

@app.get("/api/equivalents/{match\_key:path}")

def equivalents(match\_key: str):

rows = get\_key(match\_key)

if not rows:

raise HTTPException(404, "unknown formulation")

ceiling = min(r.ceiling\_per\_unit for r in rows if r.ceiling\_per\_unit)

cheapest = min(rows, key=lambda r: r.price\_per\_unit)

return JSONResponse({

"match\_key": match\_key,

"nppa\_ceiling\_per\_unit\_inr": ceiling,

"cheapest\_jap\_per\_unit\_inr": cheapest.price\_per\_unit,

"savings\_pct": round((1 - cheapest.price\_per\_unit / ceiling) \* 100, 1),

"options": \[r.as\_dict() for r in rows\],

})

@app.get("/health")

def health():

return {"status": "ok"}
